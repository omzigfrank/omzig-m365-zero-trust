       export const requireCompliantDeviceExecutor: ExecutorBundle = {
         executor: async (ctx) => {
           // Phase 1: deploy in Report-Only
           await ctx.rateLimiter.beforeWrite();
           const created = await ctx.graphClient
             .api('/identity/conditionalAccess/policies')
             .post({
               displayName: 'Omzig-CA-Require-Compliant-Device',
               state: 'enabledForReportingButNotEnforced',
               conditions: {
                 users: {
                   includeUsers: ['All'],
                   // Exclude break-glass so we never accidentally lock the emergency account out.
                   excludeGroups: ctx.facts.breakGlass.groupId ? [ctx.facts.breakGlass.groupId] : [],
                 },
                 // ... other conditions
               },
               grantControls: { operator: 'OR', builtInControls: ['compliantDevice'] },
             });
           const afterSnapshot = await ctx.graphClient
             .api(`/identity/conditionalAccess/policies/${created.id}`).get();
           return {
             beforeSnapshot: null,
             afterSnapshot,
             targetResourceId: created.id,
             phase: 'report_only_deployed',
           };
         },
         enforcePhaseExecutor: async (ctx, jobState) => {
           // Phase 2: flip to enabled
           await ctx.rateLimiter.beforeWrite();
           await ctx.graphClient
             .api(`/identity/conditionalAccess/policies/${jobState.targetResourceId}`)
             .patch({ state: 'enabled' });
           const afterSnapshot = await ctx.graphClient
             .api(`/identity/conditionalAccess/policies/${jobState.targetResourceId}`).get();
           return {
             beforeSnapshot: null,
             afterSnapshot,
             targetResourceId: jobState.targetResourceId,
             phase: 'completed',
           };
         },
         rollbackExecutor: async (ctx, beforeSnapshot, jobState) => {
           // DELETE works whether we're in Report-Only or Enforced state
           await ctx.graphClient
             .api(`/identity/conditionalAccess/policies/${jobState.targetResourceId}`).delete();
         },
         validatePrerequisites: async (ctx) => {
           // REAL fact path: facts.breakGlass (added in Plan 07-01 Task 1e). NOT facts.breakGlassGroup.

               failureReason: 'Break-glass group must exist with at least 1 member before enabling device compliance policies.',
             };
           }

         },
